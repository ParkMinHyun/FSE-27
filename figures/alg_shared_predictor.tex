\begin{algorithm}[t]
\caption{Online Update of Stage-Duration Estimates}
\label{alg:duration-update}
\footnotesize
\renewcommand{\algorithmicrequire}{\textbf{Input:}}
\renewcommand{\algorithmicensure}{\textbf{Output:}}
\algtext*{EndProcedure}\algtext*{EndIf}\algtext*{EndFor}
\begin{algorithmic}[1]
\Require Workload sequence ($\mathcal K$), measured stage durations ($p$)
\Ensure Updated base durations ($\hat b$) and shared condition factor ($\gamma$)
\State $\hat b\gets\emptyset,\quad \gamma\gets1$ \label{alg:pe:init}
\State
\Procedure{Update}{$\mathcal K,p$}
  \State $\gamma_{\mathrm{old}}\gets\gamma,\quad \mathit{ratios}\gets\emptyset$ \label{alg:pe:snapshot}
  \For{each executed stage $k\in\mathcal K$} \label{alg:pe:loop}
    \If{$\hat b(k)>0$} \label{alg:pe:observed}
      \State $\mathit{ratios}\gets\mathit{ratios}\uplus\{p(k)/\hat b(k)\}$ \label{alg:pe:ratio}
    \EndIf
    \State $\hat b(k)\gets$ cumulative average including $p(k)/\gamma_{\mathrm{old}}$ \label{alg:pe:base}
  \EndFor
  \If{$\mathit{ratios}\neq\emptyset$} \label{alg:pe:condition}
    \State $\gamma\gets\operatorname{median}(\mathit{ratios})$ \label{alg:pe:gamma}
  \EndIf
  \State \Return $(\hat b,\gamma)$ \label{alg:pe:return}
\EndProcedure
\end{algorithmic}
\end{algorithm}
